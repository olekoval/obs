1. Вивантажити дані по пацієнтах за 38 пакетом (стаціонар і амбулаторія), які одночасно отримали лікування за 17 пакетом (стаціонар і амбулаторія) за 2025 рік та 2026 рік (1-5 міс). 
2. Вивантажити дані по пацієнтах за 38 пакетом (амбулаторія), які одночасно отримали лікування за 9 пакетом за 2025 рік та 2026 рік (1-5 міс). Тут вивантажувати усе: діагностичні звіти, процедури, взаємодії і тд (все що є).
```sql
WITH aggregated_episodes AS (
    -- Крок 1: Згортаємо всі взаємодії до рівня унікального епізоду пацієнта
    SELECT 
        patient_id,
        episode_id,
        packet_number,
        MIN(starts) AS ep_start,
        -- Якщо епізод ще відкритий (ends IS NULL), обмежуємо кінцем звітного періоду
        COALESCE(MAX(ends), '2026-05-31'::timestamp) AS ep_end
    FROM analytics.rds_pmg_events_analytics
    WHERE packet_number IN ('17', '38')
      -- Фільтр звітного періоду (2025 + 5 місяців 2026)
      AND (
          (report_year = 2025) 
          OR (report_year = 2026 AND report_month BETWEEN 1 AND 5)
      )
	  AND is_correct
	  
    GROUP BY patient_id, episode_id, packet_number
),

ep_38 AS (
    -- Крок 2: Виділяємо епізоди для 38 пакета (зберігаємо episode_id)
    SELECT patient_id, episode_id, ep_start, ep_end
    FROM aggregated_episodes
    WHERE packet_number = '38'
),

ep_17 AS (
    -- Крок 3: Виділяємо епізоди для 17 пакета (тут episode_id потрібен тільки для аналізу інтервалів)
    SELECT patient_id, ep_start, ep_end
    FROM aggregated_episodes
    WHERE packet_number = '17'
),

ep_38_17 AS (
    -- Крок 4: Знаходимо перетин пацієнтів та виводимо patient_id разом із відповідним episode_id з 38 пакета
    SELECT DISTINCT
        e38.patient_id,
        e38.episode_id AS episode_id_38
    FROM ep_38 e38
    INNER JOIN ep_17 e17 ON e38.patient_id = e17.patient_id
    -- Умова одночасності (перетин часових інтервалів)
    WHERE e38.ep_start <= e17.ep_end 
      AND e17.ep_start <= e38.ep_end
),

dicts AS (
    SELECT 
        v.code, 
        v.description, 
        d.kwd_name
    FROM core.dim_rpt_dictionary_values AS v
    INNER JOIN core.dim_rpt_dictionaries AS d ON v.dictionary_id = d.id
    WHERE v.is_current = 'Y' 
      AND d.is_current = 'Y'
      AND d.kwd_name IN (
          'eHealth/encounter_admit_source', -- підстава госпіталізації
          'eHealth/encounter_discharge_disposition', -- тип виписки
          'eHealth/ICD10_AM/condition_codes', -- діагноз основний
          'eHealth/resources', -- тип події
          'eHealth/encounter_types', -- тип звернення
          'eHealth/encounter_classes',
          'eHealth/episode_types',
          'eHealth/episode_statuses',
          'POSITION', -- посади 
          'eHealth/vaccine_codes',
          'SERVICE_REQUEST_PRIORITY' -- плановий/ургентний
      )
)    

SELECT
	   e.report_year,
	   e.report_month,
	   v.residence_area,
	   e.edrpou,
	   v.kwd_name,
	   e.patient_id,
	   e.event_id,
	   e.episode_id,
	   e.packet_number,
	   e.class,
	   e.adrg,
	   e.service_number,
	   CONCAT (e.principal_diagnosis, ' - ', pd.description) AS principal_diagnosis,
	   e.add_diagnoses,
	   e.actions,
	   pr.description AS priority_code,
	   ad.description AS admission_source,
	   dm.description AS discharge_mode,
	   et.description AS event_type,
	   ent.description AS encounter_type, -- Додав розшифровку типу звернення з наявного JOIN
	   ep.description AS episode_type,
	   e.length_of_stay
  FROM analytics.rds_pmg_events_analytics AS e
	   -- Розшифровка значень кодів
		   LEFT JOIN dicts AS pd ON e.principal_diagnosis = pd.code AND pd.kwd_name = 'eHealth/ICD10_AM/condition_codes'
		   LEFT JOIN dicts AS ad ON e.admission_source = ad.code AND ad.kwd_name = 'eHealth/encounter_admit_source'
		   LEFT JOIN dicts AS dm ON e.discharge_mode = dm.code AND dm.kwd_name = 'eHealth/encounter_discharge_disposition'
		   LEFT JOIN dicts AS et ON e.event_type = et.code AND et.kwd_name = 'eHealth/resources'
		   LEFT JOIN dicts AS ent ON e.encounter_type = ent.code AND ent.kwd_name = 'eHealth/encounter_types'
		   LEFT JOIN dicts AS ep ON e.episode_type = ep.code AND ep.kwd_name = 'eHealth/episode_types'
		   LEFT JOIN dicts AS pr ON e.priority_code = pr.code AND pr.kwd_name = 'SERVICE_REQUEST_PRIORITY'
	  -- Діючи ЗОЗ
		   LEFT JOIN analytics.dwh_legal_entities_edrpou_view AS v ON v.edrpou = e.edrpou
           INNER JOIN ep_38_17 ON e.patient_id = ep_38_17.patient_id AND e.episode_id = ep_38_17.episode_id_38
WHERE e.packet_number = '38'
      -- Фільтр звітного періоду (2025 + 5 місяців 2026)
      AND (
          (e.report_year = 2025) 
          OR (e.report_year = 2026 AND e.report_month BETWEEN 1 AND 5)
      )
	  AND e.is_correct;
```
## Цікавий розрахунок кількості днів перетину конкретного епізоду
```sql
WITH target_patient AS (
    -- Визначаємо тестового пацієнта для аналізу
    SELECT '05736F6EB75D61EAEE5C8C0FAB6C8C11C82A0DF1D6C1035E89DFACF9F02F0545'::character varying AS test_id
),

patient_episodes AS (
    -- Згортаємо епізоди саме для цього пацієнта
    SELECT 
        patient_id,
        episode_id,
        packet_number,
        MIN(starts) AS ep_start,
        COALESCE(MAX(ends), '2026-05-31'::timestamp) AS ep_end
    FROM analytics.rds_pmg_events_analytics
    WHERE patient_id = (SELECT test_id FROM target_patient)
      AND packet_number IN ('17', '38')
      AND (
          (report_year = 2025) 
          OR (report_year = 2026 AND report_month BETWEEN 1 AND 5)
      )
      AND is_correct
    GROUP BY patient_id, episode_id, packet_number
)

-- Виводимо пари епізодів, які перетинаються між собою
SELECT 
    e38.patient_id,
    e38.episode_id AS episode_38_id,
    e38.ep_start AS start_38,
    e38.ep_end AS end_38,
    e17.episode_id AS episode_17_id,
    e17.ep_start AS start_17,
    e17.ep_end AS end_17,
    -- Рахуємо кількість днів спільного перетину для самоперевірки
    (LEAST(e38.ep_end, e17.ep_end)::date - GREATEST(e38.ep_start, e17.ep_start)::date) + 1 AS days_of_overlap
FROM patient_episodes e38
INNER JOIN patient_episodes e17 ON e38.patient_id = e17.patient_id
WHERE e38.packet_number = '38' 
  AND e17.packet_number = '17'
  -- Умова одночасності (перетину)
  AND e38.ep_start <= e17.ep_end 
  AND e17.ep_start <= e38.ep_end
ORDER BY e38.ep_start, e17.ep_start;
```
