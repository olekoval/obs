import pandas as pd
import numpy as np
from sqlalchemy import create_engine, text
from sqlalchemy.engine import URL
from dotenv import load_dotenv
import os


# Конфігурація підключення
load_dotenv()

url = URL.create(
    drivername=os.getenv("DB_DRIVER"),
    username=os.getenv("DB_USER"),
    host=os.getenv("DB_HOST"),
    database=os.getenv("DB_DB"),
    password=os.getenv("DB_PASSWORD"),    
)

engine = create_engine(url)

def check_connection(engine):
    """
    Функція для перевірки з'єднання з базою даних.
    """
    try:
        with engine.connect() as connection:
            # Виконуємо найпростіший запит для перевірки активності
            result = connection.execute(text("SELECT 1"))
            if result.fetchone()[0] == 1:
                print("✅ З'єднання встановлено успішно!")
                return True
        
    except Exception as e:
        print(f"❌ Помилка з'єднання: {e}")
        return False



my_sql_query = \
    """
WITH newborn_list AS (
    -- 1. Основний список новонароджених за 2025 рік по Полтавській області
    SELECT 
        person_id,
        subject_value,
        custodian AS hospital_id,
        encounter_value,
        (SELECT elem->>'valueDate' 
         FROM jsonb_array_elements(t.td_extensions) elem 
         WHERE elem->>'valueCode' = 'NEWBORN_BIRTH_DATE' 
         LIMIT 1) AS birth_date,
        (SELECT elem->>'valueString' 
         FROM jsonb_array_elements(t.td_extensions) elem 
         WHERE elem->>'valueCode' = 'NEWBORN_SEX' 
         LIMIT 1) AS gender
    FROM core.dim_rpt_composition t
    WHERE t.is_current = 'Y'
      AND t.type_coding_code = 'NEWBORN'
      AND t.status IN ('FINAL', 'AMENDED')
      AND EXISTS (
        SELECT 1
        FROM jsonb_array_elements(t.td_extensions) elem
        WHERE elem->>'valueCode' = 'NEWBORN_BIRTH_DATE'
          AND (elem->>'valueDate')::date >= '2023-01-01'::date
          AND (elem->>'valueDate')::date < '2024-01-01'::date
      )
),
measurements AS (
    -- 2. Підготовка показників (Вага та APGAR).
    SELECT 
        o.context_identifier_value AS encounter_id,
        MAX(CASE WHEN occ.code = '8339-4' 
                 THEN o.value_value_quantity_value END) AS weight_at_birth,
        MAX(CASE WHEN occ.code = 'APGAR_1' 
                 THEN o.value_value_quantity_value END) AS apgar_1min,
        MAX(CASE WHEN occ.code = 'APGAR_5' 
                 THEN o.value_value_quantity_value END) AS apgar_5min
    FROM core.dim_med_observations o
    LEFT JOIN core.dim_med_observations_code_coding occ 
        ON o.id = occ.id AND occ.is_current = 'Y'
    WHERE o.is_current = 'Y'
      AND o.context_identifier_value IN (SELECT encounter_value FROM newborn_list)
    GROUP BY o.context_identifier_value
)
-- 3. Фінальне об'єднання даних
SELECT 
    n.*,
    l.edrpou AS hospital_edrpou,
    l.kwd_name,
    l.residence_address->>'area' AS hospital_area,
    m.weight_at_birth,
    m.apgar_1min,
    m.apgar_5min
FROM newborn_list n
INNER JOIN core.dim_rpt_legal_entities l 
    ON n.hospital_id = l.id 
LEFT JOIN measurements m 
    ON n.encounter_value = m.encounter_id
WHERE l.is_current = 'Y'
  AND UPPER(l.residence_address->>'area') = 'ПОЛТАВСЬКА'
ORDER BY n.birth_date;
    """
if __name__ == "__main__":

    check_connection(engine)

    FILE = "./data/poltavska_2023_v2.csv"
    
    try:
        df = pd.read_sql(text(my_sql_query), con=engine)
        df.to_csv(FILE, index=False, encoding='utf-8-sig')
        print(f"✅ Успішно збережено {len(df)} рядків у {FILE}")
    except Exception as e:
        print(f"❌ Помилка: {e}")
    finally:
            # Повністю звільняємо ресурси пулу з'єднань
            engine.dispose()
            print("🔌 Пул з'єднань очищено.")
